```dart
// lib/features/rooms/data/datasources/rooms_remote_datasource.dart
import 'package:injectable/injectable.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

import '../../../../core/errors/domain_errors.dart';
import '../../../../core/result.dart';
import '../models/room_model.dart';

abstract class RoomsRemoteDataSource {
  Future<Result<List<RoomModel>, DomainError>> fetchRooms();

  Stream<Result<List<RoomModel>, DomainError>> watchRoomUpdates(String roomId);
}

@Injectable(as: RoomsRemoteDataSource)
class RoomsRemoteDataSourceImpl implements RoomsRemoteDataSource {
  const RoomsRemoteDataSourceImpl(this._supabase);

  final SupabaseClient _supabase;

  @override
  Future<Result<List<RoomModel>, DomainError>> fetchRooms() async {
    try {
      final response = await _supabase.from('rooms').select();

      if (response.isEmpty) {
        return const Success([]);
      }

      final rooms = (response as List)
          .map((json) => RoomModel.fromJson(json as Map<String, dynamic>))
          .toList();

      return Success(rooms);
    } catch (e) {
      return const Error(NetworkError());
    }
  }

  @override
  Stream<Result<List<RoomModel>, DomainError>> watchRoomUpdates(
    String roomId,
  ) {
    return _supabase
        .from('rooms')
        .stream(primaryKey: ['id'])
        .eq('id', roomId)
        .map((data) {
          try {
            final room = RoomModel.fromJson(data[0] as Map<String, dynamic>);
            return Success([room]);
          } catch (e) {
            return const Error(NetworkError());
          }
        })
        .handleError((_) => const Error(NetworkError()));
  }
}
```

```dart
// lib/features/rooms/data/repositories/rooms_repository_impl.dart
import 'package:injectable/injectable.dart';

import '../../../../core/errors/domain_errors.dart';
import '../../../../core/result.dart';
import '../../domain/entities/room.dart';
import '../../domain/repositories/rooms_repository.dart';
import '../datasources/rooms_local_datasource.dart';
import '../datasources/rooms_remote_datasource.dart';
import '../mappers/room_mapper.dart';

@Injectable(as: RoomsRepository)
class RoomsRepositoryImpl implements RoomsRepository {
  const RoomsRepositoryImpl({
    required RoomsRemoteDataSource remote,
    required RoomsLocalDataSource local,
  })  : _remote = remote,
        _local = local;

  final RoomsRemoteDataSource _remote;
  final RoomsLocalDataSource _local;

  @override
  Future<Result<List<Room>, DomainError>> getRooms() async {
    // Remote-first: try to fetch fresh data
    final remoteResult = await _remote.fetchRooms();

    return remoteResult.when(
      success: (models) async {
        // Cache successful result
        await _local.cacheRooms(models);
        return Success(models.map((m) => m.toDomain()).toList());
      },
      error: (error, _) async {
        // Fallback to cache if remote fails
        final cachedResult = await _local.getCachedRooms();
        return cachedResult.when(
          success: (models) => Success(models.map((m) => m.toDomain()).toList()),
          error: (_, _) => Error(error), // No cache, return remote error
        );
      },
    );
  }

  @override
  Stream<Result<List<Room>, DomainError>> watchRoom(String roomId) {
    return _remote.watchRoomUpdates(roomId).map(
      (result) => result.map(
        (models) => models.map((m) => m.toDomain()).toList(),
      ),
    );
  }
}
```

```dart
// lib/features/rooms/presentation/cubit/rooms_cubit.dart
import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:injectable/injectable.dart';

import '../../../../core/errors/domain_errors.dart';
import '../../domain/repositories/rooms_repository.dart';
import 'rooms_state.dart';

@injectable
class RoomsCubit extends Cubit<RoomsState> {
  RoomsCubit({required RoomsRepository repository})
      : _repository = repository,
        super(const RoomsInitial());

  final RoomsRepository _repository;

  Future<void> fetchRooms() async {
    emit(const RoomsLoading());
    final result = await _repository.getRooms();

    result.when(
      success: (rooms) => emit(RoomsSuccess(rooms)),
      error: (error, _) => emit(RoomsFailure(_mapErrorToMessage(error))),
    );
  }

  String _mapErrorToMessage(DomainError error) {
    return switch (error) {
      NetworkError() => 'Network error. Check your internet connection.',
      AuthError() => 'Authentication failed. Please login again.',
      CacheError() => 'Failed to load cached data.',
      _ => 'Unknown error occurred.',
    };
  }
}
```

```dart
                  },
                ),
```

```dart
import 'package:flutter_test/flutter_test.dart';

import 'package:makeup_app/core/errors/domain_errors.dart';

class MockRoomsRemoteDataSource extends Mock
    implements RoomsRemoteDataSource {}

class MockRoomsLocalDataSource extends Mock
    implements RoomsLocalDataSource {}

void main() {
  group('RoomsRepository', () {
    late MockRoomsRemoteDataSource mockRemote;
    late MockRoomsLocalDataSource mockLocal;
    late RoomsRepositoryImpl repository;

    setUp(() {
      mockRemote = MockRoomsRemoteDataSource();
      mockLocal = MockRoomsLocalDataSource();
      repository = RoomsRepositoryImpl(
        remote: mockRemote,
        local: mockLocal,
      );
    });

    final tRoomModels = [
      RoomModel(
        id: '1',
        name: 'Study Room',
        description: 'Focused study sessions',
        currentUsers: 5,
        maxUsers: 10,
        createdAt: DateTime.now(),
      ),
    ];

    test('getRooms returns rooms when remote succeeds', () async {
      // Arrange
      when(() => mockRemote.fetchRooms())
          .thenAnswer((_) async => Success(tRoomModels));
      when(() => mockLocal.cacheRooms(any()))
          .thenAnswer((_) async => Future.value());

      // Act
      final result = await repository.getRooms();

      // Assert
      expect(result, isA<Success>());
      verify(() => mockLocal.cacheRooms(tRoomModels)).called(1);
    });

    test('getRooms returns cached rooms when remote fails', () async {
      // Arrange
      when(() => mockRemote.fetchRooms())
          .thenAnswer((_) async => const Error(NetworkError()));
      when(() => mockLocal.getCachedRooms())
          .thenAnswer((_) async => Success(tRoomModels));

      // Act
      final result = await repository.getRooms();

      // Assert
      expect(result, isA<Success>());
    });

    test('getRooms returns error when both remote and cache fail', () async {
      // Arrange
      when(() => mockRemote.fetchRooms())
          .thenAnswer((_) async => const Error(NetworkError()));
      when(() => mockLocal.getCachedRooms())
          .thenAnswer((_) async => const Error(CacheError()));

      // Act
      final result = await repository.getRooms();

      // Assert
      expect(result, isA<Error>());
    });
  });
}
```

```dart
void main() {

    late MockRoomsRepository mockRepository;

    setUp(() {
      mockRepository = MockRoomsRepository();
      roomsCubit = RoomsCubit(repository: mockRepository);
    });

    final tRooms = [

        name: 'Study Room',

        currentUsers: 5,

        return roomsCubit;
      },
      act: (cubit) => cubit.fetchRooms(),
      expect: () => [
        const RoomsLoading(),
        isA<RoomsFailure>()
            .having((state) => state.message, 'message',
```
